# 공통 실행 준비

docker exec -it \
  -e CONTROL_IMAGE_ID="$(docker inspect --format '{{.Image}}' main)" \
  -e CONTROL_CONTAINER_ID="$(docker inspect --format '{{.Id}}' main)" \
  main bash

cd /workspace
source /opt/ros/humble/setup.bash
source /workspace/install/setup.bash

# ROS 로그도 호스트의 main/log 아래에 보존
export ROS_LOG_DIR="/workspace/log/ros/$(date +%Y%m%dT%H%M%S)-$$"
mkdir -p "$ROS_LOG_DIR"


# =============================
# 모터 RPM 측정
# =============================

# 터미널 1 - GNSS
ros2 launch gnss gnss.xml
  
# 터미널 2 - 모터피드백
ros2 run encoder_motor motor_rpm

# 터미널 3 - 조향피드백
ros2 run encoder_motor steer_angle

# 터미널 4 - 기록
ros2 run vehicle_recorder record --ros-args \
  -p log_root:=/workspace/log/vehicle_recorder/motor_rpm \
  -p "topics:=['/gnss/odom','/gnss/fix','/gnss/status','/motor_feedback','/steering_feedback']"

# 터미널 5,6 (옵션) - 실시간값 확인
# GNSS 속도: 단위 m/s. km/h는 ×3.6
ros2 topic echo /gnss/odom --field twist.twist.linear.x
ros2 topic echo /motor_feedback


# =============================
# 조향 anlge 측정
# =============================

# 터미널 1 - GNSS
ros2 launch gnss gnss.xml
  
# 터미널 2 - 모터피드백
ros2 run encoder_motor motor_rpm

# 터미널 3 - 조향피드백
ros2 run encoder_motor steer_angle

# 터미널 4 - 기록
ros2 run vehicle_recorder record --ros-args \
  -p log_root:=/workspace/log/vehicle_recorder/steering_angle \
  -p "topics:=['/gnss/odom','/gnss/fix','/gnss/status','/motor_feedback','/steering_feedback']"
  
# 터미널 5,6 (옵션) - 실시간값 확인
# GNSS 속도: 단위 m/s. km/h는 ×3.6
ros2 topic echo /gnss/odom --field twist.twist.linear.x
ros2 topic echo /steering_feedback


# =============================
# 제동거리 측정 — 고정 RPM STEP / GPS 실측 속도 안정화 후 자동 제동
# =============================

# 최초 1회, 소스 변경 후 빌드 (main 컨테이너 /workspace)
colcon build --packages-select manual_control trans --symlink-install
source /workspace/install/setup.bash

# 터미널 1 - GNSS
ros2 launch gnss gnss.xml

# 터미널 2 - 모터피드백
ros2 run encoder_motor motor_rpm

# 터미널 3 - 조향피드백
ros2 run encoder_motor steer_angle

# 터미널 4 - 기록 + 정지 관측 + 실차 송신
# 기존 trans 및 수동/자율 /cmd 발행기는 종료하고 이것만 실행한다.
ros2 launch manual_control brake_test_support.launch.py \
  live:=true \
  log_root:=/workspace/log/vehicle_recorder/braking

# 터미널 5 - 시험 조작 (같은 main 컨테이너의 대화형 터미널)
# 실행만으로 출발하지 않는다. 'R 시작 가능'이 뜬 뒤 R 한 번으로 시작한다.
# reference_rpm은 전진 모터 RPM 기준 명령이며 실제 GPS 차속을 의미하지 않는다.
# cmd_topic은 터미널 4와 연결하는 고정값. 아래 RPM·안정화 조건 4개를 조정한다.
ros2 run manual_control brake_step_test --ros-args \
  -p cmd_topic:=/cmd \
  -p reference_rpm:=150.0 \
  -p settle_window_s:=0.5 \
  -p settle_range_kph:=0.5 \
  -p settle_slope_kph_s:=0.2

# 제동완료 후 수동복귀시 터미널 5 result/COMPLETE 확인 -> 터미널 5 종료(Q) -> 터미널 4 종료(Ctrl+C) 순으로 진행한다.

## 터미널 5 시험 조작

키               동작

Q / Ctrl+C      정지 확인 후 시험 조작 종료
