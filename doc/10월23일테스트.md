# 공통 실행 준비

# 각 터미널에서 main 컨테이너 접속과 source를 각각 실행한다.
# 아래 colcon build만 소스 변경 후 한 번 실행하고, 다른 터미널에서도 setup.bash를 다시 source한다.
docker exec -it \
  -e CONTROL_IMAGE_ID="$(docker inspect --format '{{.Image}}' main)" \
  -e CONTROL_CONTAINER_ID="$(docker inspect --format '{{.Id}}' main)" \
  main bash

cd /workspace
source /opt/ros/humble/setup.bash
source /workspace/install/setup.bash

# ROS 로그도 호스트의 main/log 아래에 보존
export ROS_LOG_DIR="/workspace/log/ros/$(date +%Y%m%dT%H%M%S)-$$"
mkdir -p "$ROS_LOG_DIR"

# 혹시라도 소스 변경시 후 재빌드 (main 컨테이너 /workspace)
colcon build --packages-select gnss_0920 encoder_motor vehicle_recorder manual_control trans --symlink-install
source /workspace/install/setup.bash


# =============================
# 제동거리/모터가속도 측정
# =============================

# 터미널 1 - GNSS
export NTRIP_USERNAME='실제_ID'
export NTRIP_PASSWORD='실제_비밀번호'
ros2 launch gnss_0920 gnss.xml \
  ntrip_enabled:=true \
  antenna_offset_x:=0.559

# 터미널 2 - 모터피드백
ros2 run encoder_motor motor_rpm

# 터미널 3 - 조향피드백
ros2 run encoder_motor steer_angle

# 터미널 4 - 기록 + 정지 관측 + 실차 송신
# 기존 trans 및 수동/자율 /cmd 발행기는 종료하고 이것만 실행한다.
ros2 launch manual_control brake_test_support.launch.py \
  live:=true \
  log_root:=/workspace/log/vehicle_recorder/braking

# 터미널 5 - 제동시험 조작
# 실행만으로 출발하지 않는다. 'R 시작 가능'이 뜬 뒤 R 한 번으로 시작한다.
# reference_rpm은 전진 모터 RPM 기준 명령이며 실제 GPS 차속을 의미하지 않는다.
# rising_time_s=0.0: 기존 STEP. 1.0: R을 누른 시점부터 1초 동안 0 → reference_rpm 선형 상승.
# 상승 완료 후 reference_rpm을 유지하며, 그때부터 GPS 속도 안정화 구간을 수집한다.
# 상승시간은 기존 acceleration_timeout_s(기본 30초)에 포함된다.
# cmd_topic은 터미널 4와 연결하는 고정값. 아래 RPM·상승시간·안정화 조건을 조정한다.
ros2 run manual_control brake_step_test --ros-args \
  -p cmd_topic:=/cmd \
  -p reference_rpm:=150.0 \
  -p rising_time_s:=0.0 \
  -p settle_window_s:=0.5 \
  -p settle_range_kph:=0.5 \
  -p settle_slope_kph_s:=0.2

# 제동완료 후 수동복귀시 터미널 5 result/COMPLETE 확인 -> 터미널 5 종료(Q) -> 터미널 4에서 q 이후 이름 입력
# 터미널 4에서 기록 검토를 마치면 정지 관측/trans를 포함한 지원 launch도 자동 종료한다.
# 반드시 차량 정지와 터미널 5 종료를 확인한 뒤 터미널 4의 기록을 종료한다.

## 터미널 5 조작법
'''
키               동작
======================================================
R               정지 확인 후 지정 RPM 시험 시작 (rising_time_s에 따라 즉시 또는 선형 상승)
Enter / Space   모터 0·브레이크 ON
Q / Ctrl+C      정지 확인 후 시험 조작 종료
'''


# =============================
# 자율주행 테스트
# =============================

# 터미널 1 - GNSS
export NTRIP_USERNAME='실제_ID'
export NTRIP_PASSWORD='실제_비밀번호'
ros2 launch gnss_0920 gnss.xml \
  ntrip_enabled:=true \
  antenna_offset_x:=0.559

# 터미널 2 - 모터피드백
ros2 run encoder_motor motor_rpm

# 터미널 3 - 조향피드백
ros2 run encoder_motor steer_angle

# 터미널 4 - 기록
ros2 run vehicle_recorder record --ros-args \
  --params-file /workspace/src/control_manager/config/record_test.yaml

# 터미널 5 - 자율주행 조작
# --variable-speed: 지도 곡률에 따라 직선 가속/곡선 감속을 자동 계획한다.
# --cruise-speed-kph는 변속 사용 시 최고속도 상한이다.
# --cruise-speed-kph를 생략하면 기존 공통값 20km/h를 사용하므로 반드시 지정한다.
# --max-accel-kphps는 출발/재개 및 주행 중 모든 속도 증가 구간의 명령 가속도 상한((km/h)/s)이다.
# 예: 2.0이면 속도 명령이 매초 최대 2km/h 증가한다. km/s²와 다른 단위다.
# --max-decel-kphps는 일반 속도 명령 감소율 및 감속거리 계산에 사용하는 가정 감속도다.
# --max-lateral-accel-mps2는 곡선 속도 상한 계산에 사용하는 허용 횡가속도(m/s²)다.
# 가속/감속/횡가속도는 실측 보장값이 아니다. 정지/제동/입력 오류는 제한을 우회해 즉시 적용한다.
# 변속 실행에서는 최고속도·가속도·감속도·횡가속도 상한을 모두 명시해야 한다.
# 원본 mode/mission.yaml은 유지하며, 실행별 operation.yaml에 지정 속도를 보존한다.
# 실행만으로 출발하지 않는다. 일시정지 상태로 준비된 뒤 1을 눌러 출발한다.
read -r -p '속도 명령 감속도 상한((km/h)/s): ' TEST_MAX_DECEL
read -r -p '허용 횡가속도(m/s²): ' TEST_MAX_LATERAL_ACCEL
: "${TEST_MAX_DECEL:?감속도 상한을 먼저 지정하세요}"
: "${TEST_MAX_LATERAL_ACCEL:?허용 횡가속도를 먼저 지정하세요}"
ros2 run control_manager control_console --mode test --variable-speed \
  --cruise-speed-kph 20 \
  --max-accel-kphps 5 \
  --max-decel-kphps "$TEST_MAX_DECEL" \
  --max-lateral-accel-mps2 "$TEST_MAX_LATERAL_ACCEL"

# 직선에서 최고속도까지 가속하고 곡선/종점 전에 낮은 목표속도로 이행한다.
# 감속거리 계획은 지도 전체를 사용하며, 조향 lookahead는 계속 GNSS 실측 속도로 계산한다.
# response_time_s=0.5는 미측정 지연 가정이다. 실제 감속/속도 준수 여부는 GNSS로 확인한다.

# 터미널 6 - 실차 송신
ros2 run trans trans --ros-args -p cmd_topic:=/control/actuator_raw

# 주행완료후 터미널 5 종료(3) -> 터미널 4에서 q 이후 이름 입력 -> 터미널 6 종료(Ctrl+C)
# 반드시 차량 정지와 터미널 5 종료를 확인한 뒤 기록과 송신기를 종료한다.

## 터미널 5 조작법
'''
키               동작
======================================================
1 / R           출발 또는 재개
2 / P           일시정지
3 / Q / Ctrl+C  정지 확인 후 자율주행 조작 종료
'''


자율주행시 아래 사항을 확인해야 합니다.
1. 첫 GNSS COG yaw 확보에는 보통 속력 0.7m/s(2.52km/h) 이상이 필요하다.
2. 저속에서 초기화되지 않으면 정지 heading/초기 직진 bootstrap 조건을 확인한다.
3. GNSS 궤적의 경로 추종, 횡오차, 조향 포화/지연, 일정 RPM 명령에서의 실제 속도 편차, 정지/재개를 확인한다.
